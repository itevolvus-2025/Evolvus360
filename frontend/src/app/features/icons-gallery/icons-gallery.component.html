How to Add New Icons to the Store

All icon codes are stored in a single TypeScript file:

frontend/src/app/shared/ui/icon/icons.ts 
// 1. Open icons.ts and add your new icon SVG:
export const APP_ICONS: Record<string, string> = {
  // Your new icon:
  'my-custom-icon': `<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
    <circle cx="12" cy="12" r="10" />
  </svg>`,
};

// 2. In any component:
import { IconComponent } from '@shared/ui/icon';

// 3. In template:
<app-icon name="my-custom-icon" [size]="20" color="#2563EB" />
